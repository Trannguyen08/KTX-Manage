@startuml SQ_01_DangNhap
actor "Nguoi dung" as User
participant "React LoginPage" as FE
participant "authApi/httpClient" as API
participant "LoginView" as BE
database "User DB" as DB

User -> FE: Nhap username/password, chon vai tro
FE -> API: POST /api/auth/login/
API -> BE: credentials
BE -> DB: authenticate(username, password)
alt Hop le
  DB --> BE: User
  BE -> BE: Tao JWT access + refresh
  BE --> API: token, refresh, role, user info
  API --> FE: user
  FE -> FE: Luu auth_user, auth_token, auth_refresh
  FE --> User: Dieu huong /admin hoac /student
else Khong hop le
  DB --> BE: None
  BE --> API: 400 detail
  API --> FE: Loi
  FE --> User: Hien thong bao loi
end
@enduml
